  <h3>รายงานมิเตอร์ประจำเดือน</h3>
  <form id="reportForm">
    <input type="file" id="fileUpload" required>
    <button type="submit">อัพโหลด</button>
  </form>
  <div id="result"></div>

  <script>
    document.getElementById("reportForm").addEventListener("submit", function(e){
      e.preventDefault();
      const file = document.getElementById("fileUpload").files[0];
      const data = {
        action: "report",
        filename: file.name
      };

      fetch("https://script.google.com/macros/s/AKfycbzipTwJzmRZWCLpYFauUX7ZhXlveoeFey5lDwhhdRo/dev", {
        method: "POST",
        body: JSON.stringify(data)
      })
      .then(res => res.json())
      .then(data => {
        document.getElementById("result").innerText = data.status;
      });
    });
  </script>
